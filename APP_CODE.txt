import 'package:flutter/material.dart';
import 'form_field_component.dart';
import 'home_page.dart';
import 'login_page.dart';

class SignUpPage extends StatefulWidget {
  const SignUpPage({super.key});

  @override
  State<SignUpPage> createState() => SignUpPageState();
}

class SignUpPageState extends State<SignUpPage> {
  final formKey = GlobalKey<FormState>();
  final usernameController = TextEditingController();
  final emailController = TextEditingController();
  final passwordController = TextEditingController();
  final confirmPasswordController = TextEditingController();

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: const Color(0xFFFFFAFF),
      body: SafeArea(
        child: Form(
          key: formKey,
          child: ListView(
            padding: const EdgeInsets.symmetric(horizontal: 25, vertical: 35),
            children: [
              const SizedBox(height: 20),
              const Text('Sign Up', textAlign: TextAlign.center, style: TextStyle(fontSize: 36, fontWeight: FontWeight.bold)),
              const SizedBox(height: 8),
              const Text('Create your account', textAlign: TextAlign.center, style: TextStyle(fontSize: 18)),
              const SizedBox(height: 20),
              MyTextFormField(label: 'Username', icon: Icons.person_outline, controller: usernameController),
              MyTextFormField(label: 'Email', icon: Icons.mail_outline, controller: emailController,
                validator: (value) => value == null || !value.contains('@') ? 'Enter a valid email' : null),
              MyTextFormField(label: 'Password', icon: Icons.lock_outline, controller: passwordController, obscureText: true),
              MyTextFormField(label: 'Confirm Password', icon: Icons.lock_outline, controller: confirmPasswordController, obscureText: true,
                validator: (value) => value != passwordController.text ? 'Passwords do not match' : null),
              const SizedBox(height: 10),
              ElevatedButton(
                onPressed: () {
                  if (formKey.currentState!.validate()) {
                    Navigator.push(context, MaterialPageRoute(builder: (context) => const HomePage()));
                  }
                },
                style: ElevatedButton.styleFrom(backgroundColor: const Color(0xFF9215B5), foregroundColor: Colors.white, padding: const EdgeInsets.all(15)),
                child: const Text('Sign Up', style: TextStyle(fontSize: 20)),
              ),
              const SizedBox(height: 20),
              const Row(children: [Expanded(child: Divider()), Padding(padding: EdgeInsets.all(10), child: Text('Or')), Expanded(child: Divider())]),
              OutlinedButton(onPressed: () {}, child: const Text('Sign In with Google')),
              const SizedBox(height: 15),
              Row(mainAxisAlignment: MainAxisAlignment.center, children: [
                const Text('Already have an account?'),
                TextButton(onPressed: () => Navigator.push(context, MaterialPageRoute(builder: (context) => const LoginPage())), child: const Text('Login')),
              ]),
            ],
          ),
        ),
      ),
    );
  }
}

import 'package:flutter/material.dart';

class MyTextFormField extends StatelessWidget {
  final String label;
  final IconData icon;
  final TextEditingController controller;
  final FormFieldValidator<String>? validator;
  final bool obscureText;

  const MyTextFormField({
    required this.label,
    required this.icon,
    required this.controller,
    this.validator,
    this.obscureText = false,
    super.key,
  });

  @override
  Widget build(BuildContext context) {
    return Padding(
      padding: const EdgeInsets.all(10),
      child: TextFormField(
        controller: controller,
        validator: validator ?? (value) => value == null || value.isEmpty ? '$label must not be empty' : null,
        obscureText: obscureText,
        decoration: InputDecoration(
          border: OutlineInputBorder(borderRadius: BorderRadius.circular(30)),
          filled: true,
          fillColor: const Color(0xFFF1DDF8),
          hintText: label,
          prefixIcon: Icon(icon),
        ),
      ),
    );
  }
}

import 'package:flutter/material.dart';
import 'home_item.dart';
import 'home_item_component.dart';

class HomePage extends StatelessWidget {
  const HomePage({super.key});

  final List<HomeItem> items = const [
    HomeItem(name: 'Home', icon: Icons.home),
    HomeItem(name: 'Contact', icon: Icons.contact_page),
    HomeItem(name: 'Map', icon: Icons.map),
    HomeItem(name: 'Phone', icon: Icons.phone),
    HomeItem(name: 'Camera', icon: Icons.camera_alt),
    HomeItem(name: 'Settings', icon: Icons.settings),
    HomeItem(name: 'Album', icon: Icons.photo_library),
    HomeItem(name: 'WiFi', icon: Icons.wifi),
  ];

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        backgroundColor: const Color(0xFF9215B5),
        foregroundColor: Colors.white,
        centerTitle: true,
        leading: IconButton(onPressed: () {}, icon: const Icon(Icons.menu)),
        title: const Text('Flutter GridView Demo'),
        actions: [IconButton(onPressed: () {}, icon: const Icon(Icons.notifications_none))],
      ),
      body: GridView.builder(
        padding: const EdgeInsets.all(10),
        gridDelegate: const SliverGridDelegateWithFixedCrossAxisCount(
          crossAxisCount: 2,
          childAspectRatio: 1.1,
        ),
        itemBuilder: (context, index) => HomeItemWidget(item: items[index]),
        itemCount: items.length,
      ),
    );
  }
}
